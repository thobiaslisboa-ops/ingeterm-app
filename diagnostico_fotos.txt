  Future<void> _addPhoto() async {
    final source = await _pickSource();
    if (source == null) return;

    final xFile = await _picker.pickImage(source: source);
    if (xFile == null) return;

    // Verificar conexión
    final conn = await Connectivity().checkConnectivity();
    final isOnline = !conn.every((r) => r == ConnectivityResult.none);

    if (!isOnline) {
      // Sin conexión → cola de sincronización
      await ImageSyncService().addPendingImage(
        imagePath: xFile.path,
        folder: widget.offlineFolder,
        entityId: widget.offlineEntityId,
        documentPath: widget.docRef.path,
        fieldName: widget.fieldName,
      );
      _setStatus(PhotoUploadState.offlineQueued);
      return;
    }

    _setStatus(PhotoUploadState.uploading, progress: 0);

    try {
      final file = File(xFile.path);

      final compressed = await FlutterImageCompress.compressAndGetFile(
        file.absolute.path,
        '${file.parent.path}/cmp_${DateTime.now().millisecondsSinceEpoch}.jpg',
        quality: 80,
        minWidth: 1280,
        format: CompressFormat.jpeg,
      );
      final toUpload = compressed != null ? File(compressed.path) : file;

      final nombre = '${DateTime.now().millisecondsSinceEpoch}.jpg';
      final storageRef = FirebaseStorage.instance
          .ref('${widget.storagePath}/$nombre');
      final task = storageRef.putFile(
        toUpload,
        SettableMetadata(contentType: 'image/jpeg'),
      );

      task.snapshotEvents.listen((snap) {
        if (snap.totalBytes > 0 && mounted) {
          setState(() => _progress = snap.bytesTransferred / snap.totalBytes);
        }
      });

      await task.timeout(
        const Duration(seconds: 90),
        onTimeout: () {
          task.cancel();
          throw TimeoutException('Timeout subiendo imagen');
        },
      );

      final url = await storageRef.getDownloadURL();

      await widget.docRef.update({
        widget.fieldName: FieldValue.arrayUnion([url]),
      });

      try {
        if (compressed != null) await File(compressed.path).delete();
      } catch (_) {}

      _setStatus(PhotoUploadState.done);
      await Future.delayed(const Duration(seconds: 2));
      if (mounted) _setStatus(PhotoUploadState.idle);
    } catch (e) {
      _setStatus(PhotoUploadState.error);
    }
  }

  class ImageSyncService {
    static final ImageSyncService _instance = ImageSyncService._internal();
    static Database? _database;
    bool _isSyncing = false;

    // Base de datos SQLite local
    Future<Database> _getDatabase() async {

      //             documentPath, fieldName, createdAt, isCompressed)

    // Agrega imagen a la cola
    Future<void> addPendingImage({
      required String imagePath,
      required String folder,
      required String entityId,
      required String documentPath,
      String? fieldName,
    }) async {
      final db = await _getDatabase();
      await db.insert('pending_images', pending.toMap());

    }

    // Verifica conexión e intenta sincronizar
    Future<void> trySyncPendingImages() async {
      if (_isSyncing) return;

      final connectivity = Connectivity();
      final connectionStatus = await connectivity.checkConnectivity();

      // ══════════════════════════════════════════════════════════════════════
      // BUG CRÍTICO — LÍNEA 146:
      // connectivity_plus v7 devuelve List<ConnectivityResult>, NO un valor.
      // Esta comparación SIEMPRE es false → el guard nunca funciona.
      // ══════════════════════════════════════════════════════════════════════
      if (connectionStatus == ConnectivityResult.none) {  // ← SIEMPRE FALSE ⛔
        return;
      }

      await _syncPendingImages();
    }

    // Sincroniza todas las imágenes pendientes
    Future<void> _syncPendingImages() async {
      if (_isSyncing) return;
      _isSyncing = true;
      try {
        final pendingImages = await getPendingImages();
        for (final pending in pendingImages) {
          try {
            final imageFile = File(pending.imagePath);
            if (!imageFile.existsSync()) {
              await removePendingImage(pending.id!);
              continue;
            }
            // Comprimir
            final compressedFile = await FlutterImageCompress.compressAndGetFile(
              imageFile.absolute.path,
              '${imageFile.parent.path}/compressed_${DateTime.now()...}.jpg',
              quality: 80,
              format: CompressFormat.jpeg,
            );
            final fileToUpload = compressedFile != null
                ? File(compressedFile.path) : imageFile;

            // Subir a Firebase Storage
            final fileName = '${DateTime.now()...}_${basename(imageFile.path)...}';
            final ref = FirebaseStorage.instance
                .ref()
                .child('${pending.folder}/${pending.entityId}/$fileName');

            final uploadTask = ref.putFile(
              fileToUpload,
              SettableMetadata(contentType: 'image/jpeg'),
            );

            // Timeout de 90 segundos
            await uploadTask.timeout(
              const Duration(seconds: 90),
              onTimeout: () {
                uploadTask.cancel();
                throw TimeoutException('Timeout subiendo imagen', ...);
              },
            );

            final url = await ref.getDownloadURL();

            // Actualizar Firestore
            final docRef = FirebaseFirestore.instance.doc(pending.documentPath);
            if (pending.fieldName != null) {
              await docRef.update({
                pending.fieldName!: FieldValue.arrayUnion([url])
              });
            }

            await removePendingImage(pending.id!);

            // Limpiar archivo comprimido temporal
            try {
              if (compressedFile != null)
                await File(compressedFile.path).delete();
            } catch (_) {}
          } catch (_) {
            continue;  // Si falla una imagen, sigue con la siguiente
          }
        }
      } finally {
        _isSyncing = false;
      }
    }

    // Stream de conectividad (DEFINIDO pero NUNCA SUSCRITO en ningún lugar)
    Stream<bool> get connectivityStream {
      return Connectivity().onConnectivityChanged.map((results) {
        return results.isNotEmpty &&
          !results.every((r) => r == ConnectivityResult.none);
      });
    }
  }
